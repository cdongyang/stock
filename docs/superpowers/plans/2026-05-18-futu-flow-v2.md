```python
from dataclasses import dataclass
from pathlib import Path
import yaml


@dataclass
class Stock:
    code: str
    name: str


@dataclass
class ServerConfig:
    host: str
    port: int


@dataclass
class FutuConfig:
    host: str
    port: int


@dataclass
class Thresholds:
    main: float
    fat: float
    mid: float

    def tier(self, amount: float) -> str:
        if amount >= self.main:
            return "main"
        if amount >= self.fat:
            return "fat"
        if amount >= self.mid:
            return "mid"
        return "small"


@dataclass
class Config:
    stocks: list[Stock]
    server: ServerConfig
    futu: FutuConfig
    thresholds: Thresholds
    db_path: str


def load(path: str | Path | None = None) -> Config:
    if path is None:
        path = Path(__file__).parent.parent / "config" / "flow_watch.yaml"
    data = yaml.safe_load(Path(path).read_text())
    return Config(
        stocks=[Stock(**s) for s in data["stocks"]],
        server=ServerConfig(**data["server"]),
        futu=FutuConfig(**data["futu"]),
        thresholds=Thresholds(**data["thresholds"]),
        db_path=data.get("db_path", "data/flow.db"),
    )
```

```python
import os
from dataclasses import dataclass, field

import aiosqlite

_SCHEMA = """
CREATE TABLE IF NOT EXISTS capital_flow (
    id         INTEGER PRIMARY KEY,
    code       TEXT NOT NULL,
    minute     TEXT NOT NULL,
    source     TEXT NOT NULL,
    main_buy   REAL DEFAULT 0,
    main_sell  REAL DEFAULT 0,
    fat_buy    REAL DEFAULT 0,
    fat_sell   REAL DEFAULT 0,
    mid_buy    REAL DEFAULT 0,
    mid_sell   REAL DEFAULT 0,
    small_buy  REAL DEFAULT 0,
    small_sell REAL DEFAULT 0,
    UNIQUE(code, minute, source)
);
CREATE INDEX IF NOT EXISTS idx_flow ON capital_flow(code, minute, source);

CREATE TABLE IF NOT EXISTS ticks (
    id        INTEGER PRIMARY KEY,
    code      TEXT NOT NULL,
    time      TEXT NOT NULL,
    price     REAL NOT NULL,
    volume    INTEGER NOT NULL,
    amount    REAL NOT NULL,
    direction TEXT NOT NULL
);
CREATE INDEX IF NOT EXISTS idx_ticks ON ticks(code, time);
"""


@dataclass
class FlowData:
    main_buy: float = 0.0
    main_sell: float = 0.0
    fat_buy: float = 0.0
    fat_sell: float = 0.0
    mid_buy: float = 0.0
    mid_sell: float = 0.0
    small_buy: float = 0.0
    small_sell: float = 0.0


@dataclass
class TickRecord:
    code: str
    time: str
    price: float
    volume: int
    amount: float
    direction: str


class DB:
    def __init__(self, path: str):
        self.path = path
        self._conn: aiosqlite.Connection | None = None

    async def open(self):
        parent = os.path.dirname(self.path)
        if parent:
            os.makedirs(parent, exist_ok=True)
        self._conn = await aiosqlite.connect(self.path)
        await self._conn.executescript(_SCHEMA)
        await self._conn.commit()

    async def close(self):
        if self._conn:
            await self._conn.close()

    async def upsert_flow(self, code: str, minute: str, source: str, data: FlowData):
        await self._conn.execute(
            """
            INSERT INTO capital_flow
                (code, minute, source, main_buy, main_sell, fat_buy, fat_sell,
                 mid_buy, mid_sell, small_buy, small_sell)
            VALUES (?,?,?,?,?,?,?,?,?,?,?)
            ON CONFLICT(code, minute, source) DO UPDATE SET
                main_buy=excluded.main_buy,   main_sell=excluded.main_sell,
                fat_buy=excluded.fat_buy,     fat_sell=excluded.fat_sell,
                mid_buy=excluded.mid_buy,     mid_sell=excluded.mid_sell,
                small_buy=excluded.small_buy, small_sell=excluded.small_sell
            """,
            (code, minute, source,
             data.main_buy, data.main_sell,
             data.fat_buy,  data.fat_sell,
             data.mid_buy,  data.mid_sell,
             data.small_buy, data.small_sell),
        )
        await self._conn.commit()

    async def insert_tick(self, tick: TickRecord):
        await self._conn.execute(
            "INSERT INTO ticks (code, time, price, volume, amount, direction) VALUES (?,?,?,?,?,?)",
            (tick.code, tick.time, tick.price, tick.volume, tick.amount, tick.direction),
        )
        await self._conn.commit()

    async def query_flow(self, code: str, source: str, start: str, end: str) -> list[dict]:
        cols = ("minute", "main_buy", "main_sell", "fat_buy", "fat_sell",
                "mid_buy", "mid_sell", "small_buy", "small_sell")
        async with self._conn.execute(
            f"SELECT {','.join(cols)} FROM capital_flow "
            "WHERE code=? AND source=? AND minute BETWEEN ? AND ? ORDER BY minute",
            (code, source, start, end),
        ) as cur:
            return [dict(zip(cols, row)) async for row in cur]

    async def query_ticks(
        self, code: str, start: str, end: str, direction: str = "all"
    ) -> list[dict]:
        cols = ("time", "price", "volume", "amount", "direction")
        sql = "SELECT time,price,volume,amount,direction FROM ticks WHERE code=? AND time BETWEEN ? AND ?"
        params: list = [code, start, end]
        if direction in ("buy", "sell"):
            sql += " AND direction=?"
            params.append(direction)
        sql += " ORDER BY time DESC"
        async with self._conn.execute(sql, params) as cur:
            return [dict(zip(cols, row)) async for row in cur]
```

```python
from dataclasses import dataclass, field
from app.config import Thresholds

_DIR_MAP = {"Buyer": "buy", "Seller": "sell"}


@dataclass
class Classifier:
    thresholds: Thresholds
    _last: dict[str, str] = field(default_factory=dict, init=False)

    def classify(self, code: str, price: float, volume: int, ticker_direction: str) -> tuple[str, str, float]:
        """Returns (direction, tier, amount)."""
        amount = price * volume
        direction = _DIR_MAP.get(ticker_direction, self._last.get(code, "buy"))
        self._last[code] = direction
        return direction, self.thresholds.tier(amount), amount
```

```python
import asyncio
import json
import logging
from typing import Any

from fastapi import WebSocket

log = logging.getLogger(__name__)


class WSManager:
    def __init__(self):
        self._conns: list[WebSocket] = []
        self._lock = asyncio.Lock()

    async def connect(self, ws: WebSocket):
        await ws.accept()
        async with self._lock:
            self._conns.append(ws)

    async def disconnect(self, ws: WebSocket):
        async with self._lock:
            try:
                self._conns.remove(ws)
            except ValueError:
                pass

    async def broadcast(self, data: Any):
        msg = json.dumps(data, ensure_ascii=False)
        async with self._lock:
            dead: list[WebSocket] = []
            for conn in self._conns:
                try:
                    await conn.send_text(msg)
                except Exception:
                    dead.append(conn)
            for conn in dead:
                try:
                    self._conns.remove(conn)
                except ValueError:
                    pass
```

```python
import asyncio
import logging

from futu import OpenQuoteContext, RET_OK

from app.config import Config
from app.db import DB, FlowData

log = logging.getLogger(__name__)


def _f(val, default: float = 0.0) -> float:
    try:
        return float(val)
    except (TypeError, ValueError):
        return default


def _fetch_all(cfg: Config) -> list[tuple[str, str, FlowData]]:
    ctx = OpenQuoteContext(host=cfg.futu.host, port=cfg.futu.port)
    results = []
    try:
        for stock in cfg.stocks:
            ret, df = ctx.get_capital_flow(stock.code)
            if ret != RET_OK:
                log.warning("Plan A get_capital_flow %s 失败: %s", stock.code, df)
                continue
            for _, row in df.iterrows():
                t = str(row.get("capital_flow_item_time", ""))[:16]
                if not t or t == "nan":
                    continue
                results.append((
                    stock.code,
                    t,
                    FlowData(
                        main_buy=_f(row.get("main_in_flow")),
                        main_sell=_f(row.get("main_out_flow")),
                        fat_buy=_f(row.get("fat_in_flow")),
                        fat_sell=_f(row.get("fat_out_flow")),
                        mid_buy=_f(row.get("mid_in_flow")),
                        mid_sell=_f(row.get("mid_out_flow")),
                        small_buy=_f(row.get("small_in_flow")),
                        small_sell=_f(row.get("small_out_flow")),
                    ),
                ))
            log.info("Plan A 写入 %s: %d 条", stock.code, len(df))
    finally:
        ctx.close()
    return results
```

```python
import asyncio
import logging
from dataclasses import dataclass, field

from futu import OpenQuoteContext, RET_OK, SubType, TickerHandlerBase

from app.classifier import Classifier
from app.config import Config, Thresholds
from app.db import DB, FlowData, TickRecord
from app.ws import WSManager

log = logging.getLogger(__name__)


@dataclass
class _Bucket:
    data: FlowData = field(default_factory=FlowData)

    def add(self, direction: str, tier: str, amount: float):
        attr = f"{tier}_{direction}"
        setattr(self.data, attr, getattr(self.data, attr) + amount)


class _TickHandler(TickerHandlerBase):
    def __init__(self, loop: asyncio.AbstractEventLoop, queue: asyncio.Queue):
        self._loop = loop
        self._queue = queue

    def on_recv_rsp(self, rsp_pb):
        ret, df = super().on_recv_rsp(rsp_pb)
        if ret == RET_OK and df is not None and not df.empty:
            for _, row in df.iterrows():
                item = (
                    str(row.get("code", "")),
                    float(row.get("price", 0) or 0),
                    int(row.get("volume", 0) or 0),
                    str(row.get("time", ""))[:19],
                    str(row.get("ticker_direction", "") or ""),
                )
                self._loop.call_soon_threadsafe(self._queue.put_nowait, item)
        return ret, df


async def _consume(
    queue: asyncio.Queue,
    classifier: Classifier,
    db: DB,
    ws: WSManager,
):
    buckets: dict[str, tuple[str, _Bucket]] = {}
    while True:
        code, price, volume, time_str, ticker_dir = await queue.get()
        minute = time_str[:16]
        direction, tier, amount = classifier.classify(code, price, volume, ticker_dir)

        if code in buckets and buckets[code][0] != minute:
            old_minute, old_bucket = buckets.pop(code)
            await db.upsert_flow(code, old_minute, "plan_b", old_bucket.data)

        if code not in buckets:
            buckets[code] = (minute, _Bucket())

        buckets[code][1].add(direction, tier, amount)

        if tier == "main":
            tick = TickRecord(
                code=code, time=time_str, price=price,
                volume=volume, amount=amount, direction=direction,
            )
            await db.insert_tick(tick)
            await ws.broadcast({
                "type": "tick",
                "data": {
                    "code": code, "time": time_str, "price": price,
                    "volume": volume, "amount": round(amount, 2), "direction": direction,
                },
            })


def _setup_futu(
    host: str, port: int, codes: list[str],
    loop: asyncio.AbstractEventLoop, queue: asyncio.Queue,
) -> OpenQuoteContext | None:
    try:
        ctx = OpenQuoteContext(host=host, port=port)
        ctx.set_handler(_TickHandler(loop, queue))
        ret, msg = ctx.subscribe(codes, [SubType.TICKER])
        if ret != RET_OK:
            log.error("Plan B subscribe 失败: %s", msg)
            ctx.close()
            return None
        return ctx
    except Exception as e:
        log.error("Plan B futu context 创建失败: %s", e)
        return None


async def run(cfg: Config, db: DB, ws: WSManager):
    loop = asyncio.get_running_loop()
    queue: asyncio.Queue = asyncio.Queue()
    classifier = Classifier(cfg.thresholds)
    codes = [s.code for s in cfg.stocks]

    ctx = await loop.run_in_executor(
        None, _setup_futu, cfg.futu.host, cfg.futu.port, codes, loop, queue
    )
    if ctx is None:
        return

    log.info("Plan B 启动，监控: %s", codes)
    try:
        await _consume(queue, classifier, db, ws)
    finally:
        await loop.run_in_executor(None, ctx.close)
```

```python
@app.get("/api/flow/{code}")
async def get_flow(code: str, source: str = "plan_b", date: str | None = None):
    target = date or datetime.now().strftime("%Y-%m-%d")
    rows = await _db.query_flow(code, source, f"{target} 00:00", f"{target} 23:59:59")
    result = []
    cum = 0.0
    for r in rows:
        net = (r["main_buy"] + r["fat_buy"] + r["mid_buy"] + r["small_buy"]) - \
              (r["main_sell"] + r["fat_sell"] + r["mid_sell"] + r["small_sell"])
        cum += net
        result.append({
            "minute":      r["minute"],
            "main_net":    round(r["main_buy"]  - r["main_sell"],  2),
            "fat_net":     round(r["fat_buy"]   - r["fat_sell"],   2),
            "mid_net":     round(r["mid_buy"]   - r["mid_sell"],   2),
            "small_net":   round(r["small_buy"] - r["small_sell"], 2),
            "cum_net_flow": round(cum, 2),
        })
    return result


@app.get("/api/ticks/{code}")
async def get_ticks(code: str, direction: str = "all", date: str | None = None):
    target = date or datetime.now().strftime("%Y-%m-%d")
    return await _db.query_ticks(code, f"{target} 00:00", f"{target} 23:59:59", direction)
```
